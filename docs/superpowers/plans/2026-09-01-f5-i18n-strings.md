- String baru harus ditambahkan minimal ke `id.js` (target user) dan sebaiknya ke semua locale.
- AI `lang` dikirim dari `getLang()` (sudah diimplementasikan di F3/F4 plan).

### Task 2: Smoke test i18n key coverage

**Files:**
- Create: `frontend/src/lib/i18n-keys.test.js`
- Test: vitest

**Interfaces:**
- Produces: test yang memastikan tiap `t('key')` yang dipakai punya entry di locale manapun ATAU tidak error (fallback English).

- [ ] **Step 1: Write the test**

Create `frontend/src/lib/i18n-keys.test.js`:

```js
import { describe, it, expect } from 'vitest'
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const __dirname = path.dirname(fileURLToPath(import.meta.url))
const src = path.join(__dirname, '..')

function files(dir) {
  const out = []
  for (const f of fs.readdirSync(dir)) {
    const p = path.join(dir, f)
    if (fs.statSync(p).isDirectory()) out.push(...files(p))
    else if (p.endsWith('.js') || p.endsWith('.jsx')) out.push(p)
  }
  return out
}

// Collect all t('...') literal keys used in source (excluding locales/ and tests).
const used = new Set()
for (const f of files(src)) {
  if (f.includes('/locales/') || f.includes('.test.')) continue
  const text = fs.readFileSync(f, 'utf8')
  const re = /\bt\(\s*'((?:[^'\\]|\\.)*)'/g
  let m
  while ((m = re.exec(text))) used.add(m[1])
}

// Collect keys present in each locale dict.
const localeFiles = fs.readdirSync(path.join(src, 'locales')).filter(f => f.endsWith('.js'))
const dictKeys = {}
for (const lf of localeFiles) {
  const text = fs.readFileSync(path.join(src, 'locales', lf), 'utf8')
  dictKeys[lf] = new Set([...text.matchAll(/^\s*'([^']+)'\s*:/gm)].map(m => m[1]))
}

describe('i18n key coverage', () => {
  it('every t() key used in source has an entry in at least one locale (fallback = English source)', () => {
    const allKnown = new Set(Object.values(dictKeys).flatMap(s => [...s]))
    const missing = [...used].filter(k => !allKnown.has(k))
    // English source IS the key; it's always "known" as fallback, so this only flags
    // keys that are not even the English source of any locale — a genuine typo risk.
    expect(missing).toEqual([])
  })

  it('id locale covers the same key set as de locale', () => {
    expect(dictKeys['id.js'].size).toBe(dictKeys['de.js'].size)
  })
})
```

- [ ] **Step 2: Run test**

Run: `cd frontend && npm test -- i18n-keys.test.js`
Expected: PASS. Jika gagal pada "every t() key...", artinya ada `t('...')` dengan key yang tidak ada sebagai English source di locale manapun — perbaiki (biasanya typo, atau key itu sengaja English-fallback yang tidak pernah diterjemahkan; jika English source = key, key itu otomatis ada — jadi kegagalan menandakan key tidak dipakai di tempat yang konsisten).
